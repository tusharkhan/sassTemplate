create-room.js 
import { WaveNode } from '@wavenode/sdk';

const wavenode = new WaveNode(process.env.WAVENODE_SECRET_KEY);

// Create a low-latency video room with AI captions
const room = await wavenode.rooms.create({
  name: 'product-launch-live',
  region: 'auto',
  maxParticipants: 500,
  features: { recording: true, aiCaptions: true },
});

// Invite the host with a signed access token
const token = await wavenode.tokens.sign({ roomId: room.id, role: 'host' });

console.log(`Room live at ${room.joinUrl}`);
